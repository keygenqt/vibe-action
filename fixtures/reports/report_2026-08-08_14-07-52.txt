    // Backing field to store the value of type T.
    private T _value;

    // An event that notifies listeners when the value changes.
    // Action<T> is a predefined delegate that takes a parameter of type T and returns void.
    public event Action<T> ValueChanged;

// EventBus class encapsulates event management
// Uses TypeScript's class syntax with private members and type annotations
class EventBus {
    // Private handlers map stores arrays of callback functions keyed by event name
    // TypeScript's Map<string, Function[]> ensures strong typing for keys and handlers
    private handlers: Map<string, Function[]> = new Map();

    // Registers a handler function for a given event
    // Uses TypeScript's void return type to indicate no return value
    on(event: string, handler: Function): void {
        // Check if the event key does not exist in the map
        if (!this.handlers.has(event)) {
            // Initialize an empty array for this event to store handlers
            this.handlers.set(event, []);
        }
        // Retrieve the array and push the new handler
        // Non-null assertion '!' is used because we just ensured the key exists
        this.handlers.get(event)!.push(handler);
    }

    // Emits an event, calling all registered handlers with the provided data
    // Uses TypeScript's unknown type for data to allow any type safely
    emit(event: string, data: unknown): void {
        // Optional chaining '?.' ensures safe access if no handlers exist for the event
        this.handlers.get(event)?.forEach(h => h(data));
    }

Модификаторы применяются к тегам через пайп внутри фигурных скобок: {tag|modifier}.

Они позволяют преобразовывать вывод шага перед передачей дальше. Например:

- {files|join} — склеить список в одну строку
- {text|trim|upper} — обрезать пробелы и привести к верхнему регистру
- {text|split:,\s|uniq|join:\n} — разбить по запятой с пробелом, убрать дубликаты, склеить по новой строке
- {path|resolve|is_file} — превратить путь в абсолютный и проверить, что это файл
- {data|contains:ERROR} — вернуть true или false для условной логики
- {content|format:json} — сконвертировать формат между json, yaml, toml и json5
- {html|text} — извлечь чистый текст из HTML или PDF
- {num|take:5} — взять первые 5 символов или элементов списка

标签系统是所有 YAML 管线的核心。用花括号引用标签，引擎会自动解析依赖并构建 DAG。

把某一步骤的 tag 写入另一个步骤的 action 字段即可引用它的输出。例如步骤 a 的 tag 是 content，后续步骤写 {content}
就会被替换为 a 的输出。

执行顺序不由 YAML 中的排布顺序决定。引擎扫描所有 {tag} 引用，用 Kahn
算法做拓扑排序，父标签全部解析完成后才执行子步骤。循环依赖会在启动时报错。

一个步骤可以引用多个标签，例如 比较 {tag_a} 和
{tag_b}。若某步骤收到列表输入但期望字符串，会触发列表展开，对每个元素独立执行一次。用 |join 或 |join:\s
折叠列表可防止继续展开。

修饰符可自由串联，例如 {tag|trim|upper|uniq|join:\s} 会去掉空白、转大写、去重、再用空格合并。

条件分支用 when/then 矩阵实现，{tag|contains:DIRTY} 这类判断返回 true 或 false，命中的第一个 then 作为该步骤输出。

系统标签直接可用，例如 {system_clipboard} 代表剪贴板文本，{system_os} 代表当前系统，{system_dir_pwd}
代表当前目录。内置步骤也支持扫描、AST 解析、格式转换等修饰能力，完整列表见修饰符手册。
